# Ascend - Personal Growth Dashboard
## Technical Documentation & Development Reference

---

## 1. Product Overview

### Purpose
Ascend is a personal growth tracking application that helps users monitor daily habits, visualize progress, and receive AI-powered coaching through voice interaction.

### Core Features
- **Real-time score tracking** - Overall progress metric (0-100)
- **Multi-goal management** - Track coding, nutrition, screen time, etc.
- **Visual analytics** - Smooth charts and sparklines
- **Voice interface** - Natural language progress updates
- **Trend analysis** - Automatic progress calculations

## 5. Algorithm: Score Calculation

### Overall Ascend Score
```javascript
function calculateAscendScore(goals) {
  const weights = {
    coding: 0.35,    // 35% of total
    nutrition: 0.30, // 30% of total
    detox: 0.25,     // 25% of total
    bonus: 0.10      // 10% for streaks/consistency
  };
  
  let score = 0;
  
  // Weighted average of goals
  Object.keys(weights).forEach(key => {
    if (key === 'bonus') {
      score += calculateBonusPoints(goals);
    } else {
      score += goals[key].progress * weights[key];
    }
  });
  
  return Math.round(score);
}

function calculateBonusPoints(goals) {
  let bonus = 0;
  
  // Streak bonus (nutrition)
  if (goals.nutrition.streak >= 7) bonus += 3;
  else if (goals.nutrition.streak >= 4) bonus += 2;
  
  // Consistency bonus (all goals > 60%)
  const allGood = Object.values(goals)
    .every(g => g.progress >= 60);
  if (allGood) bonus += 2;
  
  return bonus;
}
```

### Trend Calculation
```javascript
function calculateTrend(history) {
  const current = history[history.length - 1];
  const previous = history[history.length - 2];
  const diff = current - previous;
  
  return {
    direction: diff >= 0 ? 'up' : 'down',
    value: Math.abs(diff).toFixed(1),
    color: diff >= 0 ? '#6ee7b7' : '#fecdd3'
  };
}
```

## 6. Chart Rendering

### Smooth Bezier Curve
```javascript
function createSmoothPath(data, width, height) {
  const normalize = (val, min, max) => 
    ((val - min) / (max - min)) * height;
  
  const max = Math.max(...data);
  const min = Math.min(...data);
  
  let path = `M 0 ${normalize(data[0], min, max)}`;
  
  for (let i = 0; i < data.length - 1; i++) {
    const x1 = (i / (data.length - 1)) * width;
    const x2 = ((i + 1) / (data.length - 1)) * width;
    const y1 = normalize(data[i], min, max);
    const y2 = normalize(data[i + 1], min, max);
    
    // Bezier control points
    const cpx1 = x1 + (x2 - x1) * 0.5;
    const cpx2 = x1 + (x2 - x1) * 0.5;
    
    path += ` C ${cpx1} ${y1}, ${cpx2} ${y2}, ${x2} ${y2}`;
  }
  
  return path;
}
```

### Sparkline Bars
```javascript
function renderSparkline(data, container) {
  container.innerHTML = '';
  
  data.forEach((value, index) => {
    const bar = document.createElement('div');
    bar.className = 'spark-bar';
    bar.style.height = `${value}%`;
    
    // Highlight last bar (today)
    if (index === data.length - 1) {
      bar.style.opacity = '0.8';
    }
    
    container.appendChild(bar);
  });
}
```

## 7. Voice Processing Flow

### Natural Language Understanding
```javascript
const voiceCommands = {
  // Completion patterns
  completed: [
    "completed {goal}",
    "finished {goal}",
    "done with {goal}",
    "checked off {goal}"
  ],
  
  // Query patterns
  status: [
    "how am I doing",
    "what's my score",
    "show my progress",
    "check my stats"
  ],
  
  // Goal-specific
  goals: [
    "coding challenge",
    "nutrition",
    "diet",
    "screen time",
    "social media"
  ]
};

function parseVoiceInput(transcript) {
  const lower = transcript.toLowerCase();
  
  // Check for completion
  if (lower.includes('completed') || lower.includes('finished')) {
    for (const goal of voiceCommands.goals) {
      if (lower.includes(goal)) {
        return {
          action: 'complete',
          target: goal,
          confidence: 0.95
        };
      }
    }
  }
  
  // Check for status query
  if (lower.includes('how') || lower.includes('score')) {
    return {
      action: 'status',
      confidence: 0.9
    };
  }
  
  return {
    action: 'unknown',
    confidence: 0.3
  };
}
```

### Response Templates
```javascript
const responses = {
```
